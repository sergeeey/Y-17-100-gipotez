"""Question 2 toy model: r+ = 2, r0 = 2, two parameters, literal Eq. (10) of Nurdin v5 vs saturation.

Basis order (s1, s2, k1, k2). rho0 = diag(q1, q2, 0, 0). A_1^dag = [[1,0],[0,0]] (k1 -> s1), A_2^dag = [[0,0],[0,1]] (k2 -> s2).
Family F_U: rho(theta) = U rho0 U^dag, U = expm(sum theta_i X_i), X_i = [[0, -B_i^dag],[B_i, 0]], B_i = A_i/2.
Checks:
 1. first-order data: d_i rho(0) = 1/2 {rho0, L_i}, L_i = [[0, A_i^dag],[A_i, 0]];
 2. F^Q = Re tr(rho0 L_a L_b);
 3. projective POVM {(s1 +- k1)/sqrt2, (s2 +- k2)/sqrt2}: all outcomes regular, F^C(0) = F^Q (point, D1);
 4. separated POVM {P+, k1 k1^dag, k2 k2^dag}: F^C at theta = t h, small t, for h generic (-> F^Q) and h on an axis (-> not F^Q);
 5. literal Eq. (10) on the kernel: no nonzero y with A_l^dag y = c A_m^dag y for all ordered pairs (exact by hand; here a
    numeric scan over the unit sphere of C^2 reports the minimum violation);
 6. negative control: replace A_2^dag by [[0,0],[1,0]] (k1 -> s2): then the fibre structure changes; report literal min violation.
usage: python -I q2_toy_r2.py OUT.json
"""

import json
import sys

import numpy as np
from scipy.linalg import expm

q = np.array([1 / 3, 2 / 3])
rho0 = np.diag([q[0], q[1], 0, 0]).astype(complex)


def model(Adag_list):
    A = [Ad.conj().T for Ad in Adag_list]
    L, X = [], []
    for Ai, Adi in zip(A, Adag_list):
        Z = np.zeros((2, 2), complex)
        L.append(np.block([[Z, Adi], [Ai, Z]]))
        B = Ai / 2
        X.append(np.block([[Z, -B.conj().T], [B, Z]]))
    return A, L, X


def rho(theta, X):
    U = expm(sum(t * Xi for t, Xi in zip(theta, X)))
    return U @ rho0 @ U.conj().T


def cfim(povm, theta, X, eps=1e-6, skip_zero=True):
    p = len(theta)
    F = np.zeros((p, p))
    r = rho(theta, X)
    for E in povm:
        pk = np.real(np.trace(r @ E))
        g = []
        for i in range(p):
            d = np.zeros(p)
            d[i] = eps
            g.append(
                (
                    np.real(np.trace(rho(theta + d, X) @ E))
                    - np.real(np.trace(rho(theta - d, X) @ E))
                )
                / (2 * eps)
            )
        g = np.array(g)
        if pk > 1e-14:
            F += np.outer(g, g) / pk
    return F


def literal_min_violation(Adag_list, n=20000, seed=0):
    rng = np.random.default_rng(seed)
    best = np.inf
    for _ in range(n):
        y = rng.normal(size=2) + 1j * rng.normal(size=2)
        y /= np.linalg.norm(y)
        a = [Ad @ y for Ad in Adag_list]
        worst = 0.0
        for l in range(len(a)):
            for m in range(len(a)):
                # min over real c of |a_l - c a_m|
                am = a[m]
                nm = np.vdot(am, am).real
                c = np.real(np.vdot(am, a[l])) / nm if nm > 1e-15 else 0.0
                worst = max(worst, np.linalg.norm(a[l] - c * am))
        best = min(best, worst)
    return float(best)


def main():
    out = sys.argv[1]
    Ad1 = np.array([[1, 0], [0, 0]], complex)
    Ad2 = np.array([[0, 0], [0, 1]], complex)
    A, L, X = model([Ad1, Ad2])
    rep = {}
    # 1. first-order data
    eps = 1e-6
    fo = 0.0
    for i in range(2):
        d = np.zeros(2)
        d[i] = eps
        drho = (rho(d, X) - rho(-d, X)) / (2 * eps)
        fo = max(fo, np.abs(drho - 0.5 * (rho0 @ L[i] + L[i] @ rho0)).max())
    rep["first_order_residual"] = float(fo)
    # 2. QFI
    FQ = np.array([[np.real(np.trace(rho0 @ L[a] @ L[b])) for b in range(2)] for a in range(2)])
    rep["FQ"] = FQ.tolist()
    # 3. projective regular POVM
    e = np.eye(4)
    vecs = [
        (e[0] + e[2]) / np.sqrt(2),
        (e[0] - e[2]) / np.sqrt(2),
        (e[1] + e[3]) / np.sqrt(2),
        (e[1] - e[3]) / np.sqrt(2),
    ]
    proj = [np.outer(v, v.conj()) for v in vecs]
    Fp = cfim(proj, np.zeros(2), X)
    rep["projective_min_prob"] = float(min(np.real(np.trace(rho0 @ E)) for E in proj))
    rep["projective_FC_minus_FQ_max"] = float(np.abs(Fp - FQ).max())
    # 4. separated POVM, ray limits
    Pplus = np.diag([1, 1, 0, 0]).astype(complex)
    sep = [Pplus, np.outer(e[2], e[2]).astype(complex), np.outer(e[3], e[3]).astype(complex)]
    rays = {}
    for name, h in {
        "generic_(1,1)": np.array([1.0, 1.0]) / np.sqrt(2),
        "axis_(1,0)": np.array([1.0, 0.0]),
    }.items():
        vals = []
        for t in (1e-2, 1e-3):
            F = cfim(sep, t * h, X, eps=1e-7)
            vals.append(
                {"t": t, "FC": F.tolist(), "max_abs_FC_minus_FQ": float(np.abs(F - FQ).max())}
            )
        rays[name] = vals
    rep["separated_ray_limits"] = rays
    rep["separated_kernel_blocks_sum_to_identity"] = True
    # 5. literal (10) on kernel
    rep["literal_min_violation_toy"] = literal_min_violation([Ad1, Ad2])
    # 6a. trivial control: A_2^dag = [[0,0],[1,0]]; satisfied only by y = k2, where BOTH images are zero
    Ad2c = np.array([[0, 0], [1, 0]], complex)
    rep["literal_min_violation_trivial_control"] = literal_min_violation([Ad1, Ad2c])
    # 6b. nontrivial control (review of 2026-11-10): A_2^dag = 2 A_1^dag; y = k1 has nonzero, real-proportional
    # images a_1 = (1,0), a_2 = (2,0), so the literal condition holds with c = 1/2 and the detector must give ~0
    Ad2n = 2 * Ad1
    rep["literal_min_violation_nontrivial_control"] = literal_min_violation([Ad1, Ad2n])
    y = np.array([1, 0], complex)
    rep["nontrivial_control_witness_images"] = [
        list(map(float, np.real(Ad1 @ y))),
        list(map(float, np.real(Ad2n @ y))),
    ]
    with open(out, "w", encoding="utf-8") as fh:
        json.dump(rep, fh, indent=1)
    print(json.dumps(rep, indent=1))


if __name__ == "__main__":
    main()
